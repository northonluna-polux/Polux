# Recurso de versión del ejecutable de Windows.
#
# Lo consume PyInstaller a través de la opción `version` del bloque EXE en
# `polux.spec`, y es lo que Windows muestra en Propiedades > Detalles.
#
# Si se cambia la versión en `version.py`, hay que actualizarla también
# aquí: el formato de este archivo no permite interpolar valores.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          '040A04B0',
          [
            StringStruct('CompanyName', 'Polux'),
            StringStruct('FileDescription',
                         'Polux - Optimizacion de rutas de reparto (VRPTW + Reglamento CE 561/2006)'),
            StringStruct('FileVersion', '1.0.0'),
            StringStruct('InternalName', 'Polux'),
            StringStruct('LegalCopyright', 'Licencia MIT'),
            StringStruct('OriginalFilename', 'Polux-1.0.0.exe'),
            StringStruct('ProductName', 'Polux'),
            StringStruct('ProductVersion', '1.0.0')
          ]
        )
      ]
    ),
    VarFileInfo([VarStruct('Translation', [0x040A, 1200])])
  ]
)
